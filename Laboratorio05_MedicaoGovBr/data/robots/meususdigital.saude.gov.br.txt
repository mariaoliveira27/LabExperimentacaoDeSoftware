    (function () {
        try {
            var params = new URLSearchParams(window.location.search || '');

            }
            var proto = window.location.protocol;
            var host = window.location.hostname || '';
            if (proto === 'capacitor:' || proto === 'ionic:' || host === 'localhost' || host === 'capacitor') {

                };
